-- 1. TABELA CENTRAL DE MOVIMENTAÇÕES (FLUXO DE CAIXA REAL)
-- Registra toda e qualquer entrada ou saída física de dinheiro na farmácia.
CREATE TABLE IF NOT EXISTS financeiro_movimentacoes (
    id_financeiro INT AUTO_INCREMENT PRIMARY KEY,
    tipo ENUM('RECEITA', 'DESPESA', 'PERDA') NOT NULL, 
    origem ENUM('VENDA', 'PERDA_ESTOQUE', 'COMPRA_PRODUTOS', 'PAGAMENTO_FUNCIONARIOS', 'CONTAS_A_PAGAR', 'AJUSTE_CAIXA') NOT NULL,
    id_referencia INT DEFAULT NULL, -- Guarda o ID da Venda, ID do Pagamento ou ID da Perda para rastreamento
    valor_bruto DECIMAL(15,2) NOT NULL DEFAULT 0.00,
    desconto DECIMAL(15,2) NOT NULL DEFAULT 0.00, -- Registra o desconto concedido de forma isolada
    acrescimo DECIMAL(15,2) NOT NULL DEFAULT 0.00,
    valor_liquido DECIMAL(15,2) NOT NULL DEFAULT 0.00, -- O valor real que alterou o caixa (Kz)
    metodo_pagamento VARCHAR(50) DEFAULT 'dinheiro', -- dinheiro, mpesa, emola, pos, misto
    id_operador INT NOT NULL, -- Quem realizou a operação (Ligado ao id_sistema do usuário)
    data_registro TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    descricao TEXT DEFAULT NULL,
    INDEX idx_tipo_origem (tipo, origem),
    INDEX idx_data (data_registro)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- 2. TABELA DE HISTÓRICO DE PAGAMENTO DE FUNCIONÁRIOS
-- Centraliza as folhas de pagamento, registrando salários, bônus e deduções.
CREATE TABLE IF NOT EXISTS financeiro_pagamentos_funcionarios (
    id_pagamento INT AUTO_INCREMENT PRIMARY KEY,
    id_funcionario INT NOT NULL, -- Referência ao id_sistema da tabela funcionarios
    mes_referencia TINYINT NOT NULL, -- Ex: 1 para Janeiro, 5 para Maio
    ano_referencia INT NOT NULL, -- Ex: 2026
    salario_base DECIMAL(15,2) NOT NULL DEFAULT 0.00,
    subsidios_bonus DECIMAL(15,2) NOT NULL DEFAULT 0.00, -- Acréscimos (Alimentação, transporte, performance)
    descontos_faltas DECIMAL(15,2) NOT NULL DEFAULT 0.00, -- Deduções por faltas ou atrasos
    descontos_impostos DECIMAL(15,2) NOT NULL DEFAULT 0.00, -- Retenções (Segurança Social / INSS / IRT)
    valor_pago_liquido DECIMAL(15,2) NOT NULL DEFAULT 0.00, -- O que o funcionário realmente recebeu
    data_pagamento DATETIME NOT NULL,
    forma_pagamento VARCHAR(50) DEFAULT 'Transferência Bancária',
    id_operador INT NOT NULL, -- Administrador que autorizou o pagamento
    status_pagamento ENUM('PAGO', 'AGUARDANDO_FUNDOS', 'CANCELADO') DEFAULT 'PAGO',
    observacoes TEXT DEFAULT NULL,
    FOREIGN KEY (id_funcionario) REFERENCES funcionarios(id_sistema) ON DELETE RESTRICT,
    INDEX idx_periodo (ano_referencia, mes_referencia)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- 3. TABELA DE CONTAS A PAGAR (CUSTOS PREVISTOS / PROVADOS)
-- Gerencia as contas futuras da farmácia (Fornecedores de medicamentos, Água, Luz, Renda, etc.)
CREATE TABLE IF NOT EXISTS financeiro_contas_pagar (
    id_conta INT AUTO_INCREMENT PRIMARY KEY,
    descricao VARCHAR(255) NOT NULL,
    categoria ENUM('FORNECEDOR', 'INFRAESTRUTURA', 'IMPOSTOS', 'MARKETING', 'OUTROS') DEFAULT 'FORNECEDOR',
    id_referencia_fornecedor INT DEFAULT NULL, -- Se for custo de fornecedor, pode amarrar ao ID dele
    valor_original DECIMAL(15,2) NOT NULL DEFAULT 0.00,
    data_vencimento DATE NOT NULL,
    data_pagamento DATE DEFAULT NULL,
    valor_pago DECIMAL(15,2) DEFAULT 0.00,
    status_conta ENUM('PENDENTE', 'PAGO_PARCIAL', 'PAGO', 'VENCIDO', 'CANCELADO') DEFAULT 'PENDENTE',
    id_operador_cadastro INT NOT NULL,
    id_operador_baixa INT DEFAULT NULL,
    data_cadastro TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_vencimento (data_vencimento),
    INDEX idx_status (status_conta)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

<?php

class PharmoraFinanceiro {
    private $pdo;

    public function __construct(PDO $pdo) {
        $this->pdo = $pdo;
    }

    // =========================================================================
    // 1. SISTEMA DE AUDITORIA
    // =========================================================================
    
    /**
     * Registra uma ação crítica no sistema para fins de auditoria.
     */
    public function registrarLogAuditoria(int $idUsuario, string $acao, string $tabela, ?int $idRegistro, string $detalhes): bool {
        $sql = "INSERT INTO auditoria_sistema (id_usuario, acao, tabela_afetada, id_registro_afetado, detalhes, ip_endereco) 
                VALUES (:id_usuario, :acao, :tabela, :id_registro, :detalhes, :ip)";
        
        $stmt = $this->pdo->prepare($sql);
        return $stmt->execute([
            ':id_usuario'  => $idUsuario,
            ':acao'        => strtoupper($acao),
            ':tabela'      => $tabela,
            ':id_registro' => $idRegistro,
            ':detalhes'    => $detalhes,
            ':ip'          => $_SERVER['REMOTE_ADDR'] ?? '127.0.0.1'
        ]);
    }

    /**
     * Recupera o histórico de auditoria com filtro opcional por usuário.
     */
    public function obterLogsAuditoria(?int $idUsuario = null, int $limite = 100): array {
        $sql = "SELECT a.*, f.id_funcionario FROM auditoria_sistema a 
                LEFT JOIN financeiro_pagamentos_funcionarios f ON a.id_usuario = f.id_funcionario ";
        
        if ($idUsuario) {
            $sql .= "WHERE a.id_usuario = :id_usuario ";
        }
        $sql .= "ORDER BY a.data_evento DESC LIMIT :limite";

        $stmt = $this->pdo->prepare($sql);
        if ($idUsuario) {
            $stmt->bindValue(':id_usuario', $idUsuario, PDO::PARAM_INT);
        }
        $stmt->bindValue(':limite', $limite, PDO::PARAM_INT);
        $stmt->execute();

        return $stmt->fetchAll(PDO::FETCH_ASSOC);
    }


    // =========================================================================
    // 2. HISTÓRICO DE VENDAS (FLUXO DE CAIXA)
    // =========================================================================
    
    /**
     * Retorna o histórico de vendas cruzando os dados reais do fluxo de caixa.
     */
    public function obterHistoricoVendas(string $dataInicio, string $dataFim): array {
        $sql = "SELECT id_financeiro, id_referencia AS id_venda, valor_bruto, desconto, acrescimo, 
                       valor_liquido, metodo_pagamento, id_operador, data_registro, descricao
                FROM financeiro_movimentacoes 
                WHERE tipo = 'RECEITA' AND origem = 'VENDA' 
                AND data_registro BETWEEN :inicio AND :fim
                ORDER BY data_registro DESC";

        $stmt = $this->pdo->prepare($sql);
        $stmt->execute([
            ':inicio' => $dataInicio . ' 00:00:00',
            ':fim'    => $dataFim . ' 23:59:59'
        ]);

        return $stmt->fetchAll(PDO::FETCH_ASSOC);
    }


    // =========================================================================
    // 3. GESTÃO DE CONTAS A PAGAR
    // =========================================================================
    
    /**
     * Lista as contas cadastradas com base no status (PENDENTE, PAGO, VENCIDO, etc).
     */
    public function listarContasAPagar(?string $status = null): array {
        $sql = "SELECT * FROM financeiro_contas_pagar ";
        if ($status) {
            $sql .= "WHERE status_conta = :status ";
        }
        $sql .= "ORDER BY data_vencimento ASC";

        $stmt = $this->pdo->prepare($sql);
        if ($status) {
            $stmt->bindValue(':status', strtoupper($status), PDO::PARAM_STR);
        }
        $stmt->execute();

        return $stmt->fetchAll(PDO::FETCH_ASSOC);
    }

    /**
     * Efetua a baixa (pagamento) de uma conta e gera automaticamente a movimentação de saída.
     */
    public function pagarConta(int $idConta, float $valorPago, int $idOperador, string $metodoPagamento = 'Transferência Bancária'): bool {
        try {
            $this->pdo->beginTransaction();

            // 1. Buscar dados da conta para validação
            $stmt = $this->pdo->prepare("SELECT * FROM financeiro_contas_pagar WHERE id_conta = ? FOR UPDATE");
            $stmt->execute([$idConta]);
            $conta = $stmt->fetch(PDO::FETCH_ASSOC);

            if (!$conta || $conta['status_conta'] === 'PAGO') {
                throw new Exception("Conta inválida ou já liquidada.");
            }

            // 2. Atualizar a tabela de Contas a Pagar
            $sqlAtualizaConta = "UPDATE financeiro_contas_pagar 
                                 SET status_conta = 'PAGO', data_pagamento = CURDATE(), valor_pago = :valor_pago, id_operador_baixa = :operador 
                                 WHERE id_conta = :id_conta";
            $stmtUpdate = $this->pdo->prepare($sqlAtualizaConta);
            $stmtUpdate->execute([
                ':valor_pago' => $valorPago,
                ':operador'   => $idOperador,
                ':id_conta'   => $idConta
            ]);

            // 3. Lançar a saída física de dinheiro na tabela central de movimentações
            $sqlMovimentacao = "INSERT INTO financeiro_movimentacoes 
                                (tipo, origem, id_referencia, valor_bruto, valor_liquido, metodo_pagamento, id_operador, descricao) 
                                VALUES ('DESPESA', 'CONTAS_A_PAGAR', :id_ref, :valor, :valor, :metodo, :operador, :descricao)";
            $stmtMov = $this->pdo->prepare($sqlMovimentacao);
            $stmtMov->execute([
                ':id_ref'    => $idConta,
                ':valor'     => $valorPago,
                ':metodo'    => $metodoPagamento,
                ':operador'  => $idOperador,
                ':descricao' => "Liquidação da conta: " . $conta['descricao']
            ]);

            // 4. Registrar na Auditoria
            $this->registrarLogAuditoria(
                $idOperador, 
                'BAIXA_CONTA', 
                'financeiro_contas_pagar', 
                $idConta, 
                "Conta '{$conta['descricao']}' paga no valor de Kz $valorPago"
            );

            $this->pdo->commit();
            return true;

        } catch (Exception $e) {
            $this->pdo->rollBack();
            error_log("Erro ao pagar conta: " . $e->getMessage());
            return false;
        }
    }


    // =========================================================================
    // 4. GESTÃO DE PERDAS DE ESTOQUE
    // =========================================================================
    
    /**
     * Registra a perda de um produto, detalha o motivo e abate o valor no fluxo financeiro.
     */
    public function registrarPerdaEstoque(int $idProduto, int $quantidade, string $motivo, float $valorPrejuizoTotal, int $idOperador): bool {
        try {
            $this->pdo->beginTransaction();

            // 1. Inserir o detalhe na tabela de perdas
            $sqlPerda = "INSERT INTO produtos_perdas (id_produto, quantidade, motivo, valor_prejuizo, id_operador) 
                         VALUES (:id_produto, :qtd, :motivo, :prejuizo, :operador)";
            $stmtPerda = $this->pdo->prepare($sqlPerda);
            $stmtPerda->execute([
                ':id_produto' => $idProduto,
                ':qtd'        => $quantidade,
                ':motivo'     => $motivo,
                ':prejuizo'   => $valorPrejuizoTotal,
                ':operador'   => $idOperador
            ]);
            
            $idPerdaGerada = $this->pdo->lastInsertId();

            // 2. Lançar o impacto negativo na tabela de movimentações financeiras
            $sqlMovimentacao = "INSERT INTO financeiro_movimentacoes 
                                (tipo, origem, id_referencia, valor_bruto, valor_liquido, id_operador, descricao) 
                                VALUES ('PERDA', 'PERDA_ESTOQUE', :id_ref, :valor, :valor, :operador, :descricao)";
            $stmtMov = $this->pdo->prepare($sqlMovimentacao);
            $stmtMov->execute([
                ':id_ref'    => $idPerdaGerada,
                ':valor'     => $valorPrejuizoTotal,
                ':operador'  => $idOperador,
                ':descricao' => "Perda de Estoque - Qtd: $quantidade | Motivo: $motivo"
            ]);

            // 3. Auditar a ação
            $this->registrarLogAuditoria(
                $idOperador, 
                'REGISTRO_PERDA', 
                'produtos_perdas', 
                $idPerdaGerada, 
                "Produto ID $idProduto. Qtd: $quantidade. Prejuízo avaliado em Kz $valorPrejuizoTotal"
            );

            $this->pdo->commit();
            return true;

        } catch (Exception $e) {
            $this->pdo->rollBack();
            error_log("Erro ao registrar perda: " . $e->getMessage());
            return false;
        }
    }

    /**
     * Retorna o relatório completo de perdas registradas em um período.
     */
    public function obterHistoricoPerdas(string $dataInicio, string $dataFim): array {
        $sql = "SELECT p.*, m.valor_liquido 
                FROM produtos_perdas p
                INNER JOIN financeiro_movimentacoes m ON p.id_perda = m.id_referencia 
                WHERE m.origem = 'PERDA_ESTOQUE' 
                AND p.data_registro BETWEEN :inicio AND :fim
                ORDER BY p.data_registro DESC";

        $stmt = $this->pdo->prepare($sql);
        $stmt->execute([
            ':inicio' => $dataInicio . ' 00:00:00',
            ':fim'    => $dataFim . ' 23:59:59'
        ]);

        return $stmt->fetchAll(PDO::FETCH_ASSOC);
    }
}
